var SPREADSHEET_ID = "1LBQbIgLWXR6e_vUEBjiomtV9XJloUswzfv70ZE9uPL0"; var SHEET_NAME = "DATA_INVOICE"; var FOLDER_NAME = "Dokumentasi_Invoice_Sorong"; function getTargetFolder() { var folders = DriveApp.getFoldersByName(FOLDER_NAME); if (folders.hasNext()) { return folders.next(); } else { return DriveApp.createFolder(FOLDER_NAME); } } function getTargetSheet() { var ss = SpreadsheetApp.openById(SPREADSHEET_ID); var sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { throw new Error("Tab " + SHEET_NAME + " tidak ditemukan di Google Spreadsheet."); } return sheet; } function doPost(e) { try { var rawData = e.postData.contents; var data = JSON.parse(rawData); var sheet = getTargetSheet(); var rows = sheet.getDataRange().getValues(); var noBillingTarget = String(data.NoBilling).trim(); var targetRow = -1; for (var i = 1; i < rows.length; i++) { if (String(rows[i][0]).trim() === noBillingTarget) { targetRow = i + 1; break; } } if (targetRow !== -1) { if (data.TglKirim) sheet.getRange(targetRow, 11).setValue(data.TglKirim); // Kolom K: TGL PENGIRIMAN if (data.DateReceive) sheet.getRange(targetRow, 12).setValue(data.DateReceive); // Kolom L: DATE RECEIVE // Proses penyimpanan foto/gambar ke Google Drive & tautannya ke Kolom M (13) if (data.fileData && data.fileName) { var folder = getTargetFolder(); var decoded = Utilities.base64Decode(data.fileData); var blob = Utilities.newBlob(decoded, data.mimeType || 'image/jpeg', data.fileName); var file = folder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); var fileUrl = file.getUrl(); sheet.getRange(targetRow, 13).setValue(fileUrl); // Simpan URL ke Kolom M } if (data.Dokumentasi) sheet.getRange(targetRow, 14).setValue(data.Dokumentasi); // Kolom N: STATUS PENGIRIMAN return ContentService.createTextOutput(JSON.stringify({"status": "success", "row": targetRow})) .setMimeType(ContentService.MimeType.JSON); } else { return ContentService.createTextOutput(JSON.stringify({"status": "error", "message": "No Billing tidak ditemukan"})) .setMimeType(ContentService.MimeType.JSON); } } catch(err) { return ContentService.createTextOutput(JSON.stringify({"status": "error", "message": err.toString()})) .setMimeType(ContentService.MimeType.JSON); } } function doGet(e) { try { var sheet = getTargetSheet(); var rows = sheet.getDataRange().getDisplayValues(); var dataList = []; for (var i = 1; i < rows.length; i++) { var row = rows[i]; if (row[0]) { var rawKomersil = String(row[5] || "0").replace(/[^\d]/g, ''); var numKomersil = parseFloat(rawKomersil) || 0; dataList.push({ "NoBilling": String(row[0] || "").trim(), "BillingDate": String(row[1] || "").trim(), "ShipTo": String(row[3] || ""), "Kota": String(row[4] || ""), "Komersil": numKomersil, "PPn": Number(row[6] || 0), "Salesman": String(row[7] || ""), "Barang": String(row[8] || ""), "Expedisi": String(row[9] || ""), "TglKirim": String(row[10] || "").trim(), "DateReceive": String(row[11] || "").trim(), "SP": String(row[12] || ""), // Kolom M: Tautan Foto SP "Dokumentasi": String(row[13] || "BELUM DIKIRIM") }); } } return ContentService.createTextOutput(JSON.stringify(dataList)) .setMimeType(ContentService.MimeType.JSON); } catch(err) { return ContentService.createTextOutput(JSON.stringify({"status": "error", "message": err.toString()})) .setMimeType(ContentService.MimeType.JSON); } }
